  if exists(select 1 from private.eu_news_source_provenance where source_id=old.id) then
    raise exception using errcode='42501',message='immutable_policy_source'; end if;
  if tg_op='DELETE' then return old; end if; return new;
end $$;

create function private.eu_news_claim_guard() returns trigger
language plpgsql security definer set search_path = '' as $$
begin
  if not exists(select 1 from private.eu_news_claim_origins where claim_id=old.id) then
    if tg_op='DELETE' then return old; end if; return new;
  end if;
  if tg_op='DELETE' then raise exception using errcode='42501',message='immutable_policy_claim'; end if;
  if (to_jsonb(new)-array['status','approved_by_m_uid','updated_at']) is distinct from (to_jsonb(old)-array['status','approved_by_m_uid','updated_at']) then
    raise exception using errcode='42501',message='immutable_policy_claim';
  end if;
  if not exists(select 1 from private.eu_news_claim_reviews r where r.claim_id=old.id and r.from_status=old.status
    and r.to_status=new.status and r.actor_m_uid=private.eu_news_actor() and r.transaction_id=txid_current()
    and (new.approved_by_m_uid is not distinct from case when new.status='approved' then r.actor_m_uid else null end)) then
    raise exception using errcode='42501',message='atomic_independent_review_required';
  end if;
  return new;
end $$;
create trigger eu_news_claim_guard before update or delete on public.eu_claims for each row execute function private.eu_news_claim_guard();
